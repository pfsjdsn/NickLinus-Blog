```js
this.arr1 = [{ name: "张三" }, { name: "李四" }];
this.arr2 = [
  { study: true, age: 18 },
  { study: true, age: 99 },
];
this.arr1 = this.arr1.map((item, index) => {
  const person = this.arr2.find((value, pIndex) => pIndex === index);
  return { ...item, age: person ? person.age : "" };
});
console.log(this.arr1);
/**
 * [{age: 18,name: "张三"},{age: 99,name: "李四"}]
 */
```
