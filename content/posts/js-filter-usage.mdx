```javascript
let messageId =  '1'
let list = [{name: 'a', id: '1'}, {name: 'b', id: '2'}, {name: 'c', id: '3'}]
list  = list.filter((val, i) => {
return val.id !== messageId
})
console.log(list) // [{name: 'b', id: '2'}, {name: 'c', id: '3'}]
```
