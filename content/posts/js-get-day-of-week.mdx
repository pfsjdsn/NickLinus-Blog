```js
      getWeek(){
        let date = new Date();
        let options = { weekday: "long" };
        let day = date.toLocaleString("zh-CN", options); // 中文：星期二
        // let day = date.toLocaleString("en-US", options) // 英文：Tuesday
        return day;
      }
```
