以下代码是在uniapp环境中

```js
onLoad() {
      let actualDate = '2024-03-12'
      let scheduleDate = '2024-03-06'
      const days = this.dateDifference(actualDate, scheduleDate);
      console.log(days); // 输出 6
  },
methods:{
    dateDifference(date1, date2) {
      // 转换为Date对象
      const d1 = new Date(date1);
      const d2 = new Date(date2);
      // 获取日期差
      const diff = d1 - d2;
      // 计算相差的天数
      const days = Math.round(diff / (1000 * 60 * 60 * 24));
      return days;
    }
}

```
