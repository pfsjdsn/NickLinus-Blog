```js
//  获取分钟/小时差
export function getMinuteDifference(start) {
  let startData = new Date(start);
  let stopData = new Date();
  let startTime = startData.getTime();
  let stopTime = stopData.getTime();
  let ctiTime = Number(stopTime) - Number(startTime);
  let secondTime = "";
  if (ctiTime / 1000 / 60 / 60 > 24) {
    secondTime =
      Math.floor(ctiTime / 1000 / 60 / 60 / 24) === 0
        ? 1 + "天前"
        : Math.floor(ctiTime / 1000 / 60 / 60 / 24) + "天前";
  } else if (ctiTime / 1000 / 60 > 60) {
    secondTime =
      Math.floor(ctiTime / 1000 / 60 / 60) === 0
        ? 1 + "小时前"
        : Math.floor(ctiTime / 1000 / 60 / 60) + "小时前";
  } else {
    secondTime =
      Math.floor(ctiTime / 1000 / 60) === 0
        ? 1 + "分钟前"
        : Math.floor(ctiTime / 1000 / 60) + "分钟前";
  }
  return secondTime;
}
```
