filters: {
    // 只显示到两位小数点
    rounding (value) {
      // console.log(value);
      return value.toFixed(2)
      // console.log(value);
    },
    // 以百分比显示
    percent (value, fmt) {
      if (value !== '') {
        return `${((value || 0) * 100).toFixed(fmt || 2)}%`
      } else {
        return value
      }
    },
}
