```javascript
 data: {
    coverObj: {},
  }
onload() {
let {coverObj} = this.data
	 coverObj.age =  18
	 coverObj.name =  jack
	  this.setData({
      coverObj
    })
    console.log(this.data.coverObj) // {age： 18，name： 'jack' }
}
```
