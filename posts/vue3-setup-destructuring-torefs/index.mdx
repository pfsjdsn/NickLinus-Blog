```vue
<template>
  <h1>用户年龄：{{ age }}</h1>
  <h1>用户名称：{{ name }}</h1>
</template>
```

```vue
<script>
import { reactive } from 'vue'
export default {
  data () {
    return {
    }
  },
  methods: {
  },
  setup () {
    const user = reactive({
      age: '18',
      name: 'abc'
    })
    return {  ...toRefs(user)  }
  },
</script>
```
